# RAG Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the existing `notes_fts`/`notes_meta` schema with a chunk-based RAG schema and wire up FTS-only search as the foundation for hybrid search.

**Architecture:** New `chunks` / `file_cache` / `fts_chunks` / `vec_chunks` tables replace the old two-table design. A new `chunker.rs` splits Markdown by headers/paragraphs. A new `embedder.rs` wraps `ort` + `tokenizers` for ONNX inference. The existing `indexer.rs`, `db.rs`, and `search.rs` are rewritten to use the new schema. The CLI gains `--mode`, `dredge`, and `setup` subcommands. Vector search is behind an optional embedder — FTS always works, vector gracefully degrades when the model is missing.

**Tech Stack:** `rusqlite` (bundled), `sqlite-vec` 0.1.x, `ort` 2.x, `tokenizers` 0.19.x, `ndarray` 0.15.x, `indicatif` 0.17.x, `tokio` 1.x (multi-thread), `rayon` 1.x

```rust
use crate::models::{Chunk, ChunkSearchResult, SearchMode, VaultStats};
use rusqlite::{params, Connection, OpenFlags, Result as SqliteResult};
use std::cell::RefCell;
use std::path::{Path, PathBuf};
use thiserror::Error;

#[derive(Error, Debug)]
pub enum DbError {
    #[error("SQLite error: {0}")]
    Sqlite(#[from] rusqlite::Error),
    #[error("Note not found: {0}")]
    NotFound(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

/// Manages the SQLite database — write connection for indexer,
/// read-only connection for search (WAL allows concurrent readers).
pub struct NoteDatabase {
    pub write_conn: RefCell<Connection>,
}

impl NoteDatabase {
    pub fn open<P: AsRef<Path>>(path: P) -> Result<Self, DbError> {
        let is_fresh = !path.as_ref().exists();
        let conn = Connection::open(&path)?;
        conn.pragma_update(None, "journal_mode", "WAL")?;
        let db = Self { write_conn: RefCell::new(conn) };
        db.migrate()?;
        #[cfg(unix)]
        if is_fresh {
            use std::os::unix::fs::PermissionsExt;
            let _ = std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o600));
            let base = path.as_ref().to_string_lossy();
            for suffix in ["-wal", "-shm"] {
                let companion = PathBuf::from(format!("{}{}", base, suffix));
                if companion.exists() {
                    let _ = std::fs::set_permissions(&companion, std::fs::Permissions::from_mode(0o600));
                }
            }
        }
        Ok(db)
    }

    pub fn open_in_memory() -> Result<Self, DbError> {
        let conn = Connection::open_in_memory()?;
        let db = Self { write_conn: RefCell::new(conn) };
        db.migrate()?;
        Ok(db)
    }

    /// Open a read-only connection to an existing DB (for MCP search handler).
    pub fn open_readonly<P: AsRef<Path>>(path: P) -> Result<Connection, DbError> {
        let conn = Connection::open_with_flags(path, OpenFlags::SQLITE_OPEN_READ_ONLY)?;
        Ok(conn)
    }

    fn migrate(&self) -> Result<(), DbError> {
        let conn = self.write_conn.borrow();
        let version: i64 = conn.query_row("PRAGMA user_version", [], |r| r.get(0)).unwrap_or(0);

        if version < 2 {
            // Drop v1 tables if present
            conn.execute_batch("
                DROP TABLE IF EXISTS notes_fts;
                DROP TABLE IF EXISTS notes_meta;
            ")?;
            self.create_schema(&conn)?;
            conn.execute_batch("PRAGMA user_version = 2")?;
        }
        Ok(())
    }

    fn create_schema(&self, conn: &Connection) -> SqliteResult<()> {
        conn.execute_batch("
            CREATE TABLE IF NOT EXISTS file_cache (
                path     TEXT PRIMARY KEY,
                hash     TEXT NOT NULL,
                mtime    INTEGER NOT NULL,
                model_id TEXT NOT NULL
            );

            CREATE TABLE IF NOT EXISTS chunks (
                id           INTEGER PRIMARY KEY,
                file_path    TEXT NOT NULL,
                chunk_index  INTEGER NOT NULL,
                parent_header TEXT,
                content      TEXT NOT NULL
            );
            CREATE INDEX IF NOT EXISTS idx_chunks_file_path ON chunks(file_path);

            CREATE VIRTUAL TABLE IF NOT EXISTS fts_chunks USING fts5(
                content,
                content='chunks',
                content_rowid='id',
                tokenize='unicode61 remove_diacritics 0'
            );

            CREATE VIRTUAL TABLE IF NOT EXISTS vec_chunks USING vec0(
                chunk_id  INTEGER PRIMARY KEY,
                embedding FLOAT[1024]
            );
        ")?;
        Ok(())
    }

    /// Insert a batch of chunks for a file. Caller must have deleted old chunks first.
    pub fn insert_chunks(&self, chunks: &[Chunk]) -> Result<Vec<i64>, DbError> {
        let mut conn = self.write_conn.borrow_mut();
        let tx = conn.transaction()?;
        let mut ids = Vec::with_capacity(chunks.len());
        for chunk in chunks {
            tx.execute(
                "INSERT INTO chunks (file_path, chunk_index, parent_header, content) VALUES (?1,?2,?3,?4)",
                params![chunk.file_path, chunk.chunk_index, chunk.parent_header, chunk.content],
            )?;
            let id = tx.last_insert_rowid();
            // FTS insert (contentless — must mirror content)
            tx.execute(
                "INSERT INTO fts_chunks(rowid, content) VALUES (?1, ?2)",
                params![id, chunk.content],
            )?;
            ids.push(id);
        }
        tx.commit()?;
        Ok(ids)
    }

    /// Insert embeddings for a batch of (chunk_id, embedding) pairs.
    pub fn insert_embeddings(&self, pairs: &[(i64, Vec<f32>)]) -> Result<(), DbError> {
        let mut conn = self.write_conn.borrow_mut();
        let tx = conn.transaction()?;
        for (chunk_id, embedding) in pairs {
            let blob: Vec<u8> = embedding.iter()
                .flat_map(|f| f.to_le_bytes())
                .collect();
            tx.execute(
                "INSERT INTO vec_chunks(chunk_id, embedding) VALUES (?1, ?2)",
                params![chunk_id, blob],
            )?;
        }
        tx.commit()?;
        Ok(())
    }

    /// Delete all chunks (and their FTS/vec entries) for a file path.
    /// Uses the special contentless-FTS delete syntax for fts_chunks.
    pub fn delete_chunks_for_file(&self, file_path: &str) -> Result<(), DbError> {
        let mut conn = self.write_conn.borrow_mut();
        let tx = conn.transaction()?;

        // Collect chunk ids and content for FTS delete
        let rows: Vec<(i64, String)> = {
            let mut stmt = tx.prepare(
                "SELECT id, content FROM chunks WHERE file_path = ?1"
            )?;
            stmt.query_map([file_path], |r| Ok((r.get(0)?, r.get(1)?)))?
                .collect::<SqliteResult<Vec<_>>>()?
        };

        for (id, content) in &rows {
            // Contentless FTS5 delete requires re-supplying the content
            tx.execute(
                "INSERT INTO fts_chunks(fts_chunks, rowid, content) VALUES('delete', ?1, ?2)",
                params![id, content],
            )?;
            // vec_chunks delete by chunk_id
            tx.execute("DELETE FROM vec_chunks WHERE chunk_id = ?1", [id])?;
        }

        tx.execute("DELETE FROM chunks WHERE file_path = ?1", [file_path])?;
        tx.commit()?;
        Ok(())
    }

    /// Upsert file_cache entry.
    pub fn upsert_file_cache(
        &self,
        path: &str,
        hash: &str,
        mtime: i64,
        model_id: &str,
    ) -> Result<(), DbError> {
        self.write_conn.borrow().execute(
            "INSERT INTO file_cache (path, hash, mtime, model_id)
             VALUES (?1,?2,?3,?4)
             ON CONFLICT(path) DO UPDATE SET
                 hash=excluded.hash, mtime=excluded.mtime, model_id=excluded.model_id",
            params![path, hash, mtime, model_id],
        )?;
        Ok(())
    }

    /// Returns the stored hash for a file, or None if not cached.
    pub fn cached_hash(&self, path: &str) -> Result<Option<String>, DbError> {
        let conn = self.write_conn.borrow();
        match conn.query_row(
            "SELECT hash FROM file_cache WHERE path = ?1",
            [path],
            |r| r.get(0),
        ) {
            Ok(h) => Ok(Some(h)),
            Err(rusqlite::Error::QueryReturnedNoRows) => Ok(None),
            Err(e) => Err(DbError::Sqlite(e)),
        }
    }

    /// Delete file_cache entry for a file.
    pub fn delete_file_cache(&self, path: &str) -> Result<(), DbError> {
        self.write_conn.borrow().execute(
            "DELETE FROM file_cache WHERE path = ?1",
            [path],
        )?;
        Ok(())
    }

    /// FTS search on fts_chunks. Returns (chunk_id, score) pairs.
    pub fn fts_search(&self, fts5_query: &str, limit: usize) -> Result<Vec<(i64, f64)>, DbError> {
        let conn = self.write_conn.borrow();
        let mut stmt = conn.prepare(
            "SELECT rowid, rank FROM fts_chunks WHERE fts_chunks MATCH ?1 ORDER BY rank LIMIT ?2"
        )?;
        let rows = stmt.query_map(params![fts5_query, limit as i64], |r| {
            Ok((r.get::<_, i64>(0)?, r.get::<_, f64>(1)?))
        })?;
        rows.collect::<SqliteResult<Vec<_>>>().map_err(DbError::Sqlite)
    }

    /// Vector KNN search on vec_chunks. Returns (chunk_id, distance) pairs.
    pub fn vec_search(&self, embedding: &[f32], limit: usize) -> Result<Vec<(i64, f64)>, DbError> {
        let conn = self.write_conn.borrow();
        let blob: Vec<u8> = embedding.iter().flat_map(|f| f.to_le_bytes()).collect();
        let mut stmt = conn.prepare(
            "SELECT chunk_id, distance FROM vec_chunks
             WHERE embedding MATCH ?1 AND k = ?2
             ORDER BY distance"
        )?;
        let rows = stmt.query_map(params![blob, limit as i64], |r| {
            Ok((r.get::<_, i64>(0)?, r.get::<_, f64>(1)?))
        })?;
        rows.collect::<SqliteResult<Vec<_>>>().map_err(DbError::Sqlite)
    }

    /// Fetch chunks by ids, preserving order.
    pub fn get_chunks_by_ids(&self, ids: &[i64]) -> Result<Vec<Chunk>, DbError> {
        if ids.is_empty() { return Ok(vec![]); }
        let conn = self.write_conn.borrow();
        let placeholders: String = ids.iter().enumerate()
            .map(|(i, _)| format!("?{}", i + 1))
            .collect::<Vec<_>>()
            .join(",");
        let sql = format!(
            "SELECT id, file_path, chunk_index, parent_header, content FROM chunks WHERE id IN ({})",
            placeholders
        );
        let mut stmt = conn.prepare(&sql)?;
        let params_vec: Vec<&dyn rusqlite::ToSql> = ids.iter()
            .map(|id| id as &dyn rusqlite::ToSql)
            .collect();
        let rows = stmt.query_map(params_vec.as_slice(), |r| {
            Ok(Chunk {
                id: Some(r.get(0)?),
                file_path: r.get(1)?,
                chunk_index: r.get(2)?,
                parent_header: r.get(3)?,
                content: r.get(4)?,
            })
        })?;
        rows.collect::<SqliteResult<Vec<_>>>().map_err(DbError::Sqlite)
    }

    /// Fetch chunks surrounding a given chunk_id (for MCP get_surrounding_context).
    pub fn get_surrounding_chunks(&self, chunk_id: i64, window: usize) -> Result<Vec<Chunk>, DbError> {
        let conn = self.write_conn.borrow();
        let file_path: String = conn.query_row(
            "SELECT file_path FROM chunks WHERE id = ?1", [chunk_id], |r| r.get(0)
        ).map_err(|e| match e {
            rusqlite::Error::QueryReturnedNoRows => DbError::NotFound(chunk_id.to_string()),
            other => DbError::Sqlite(other),
        })?;
        let chunk_index: i64 = conn.query_row(
            "SELECT chunk_index FROM chunks WHERE id = ?1", [chunk_id], |r| r.get(0)
        )?;
        let w = window as i64;
        let mut stmt = conn.prepare(
            "SELECT id, file_path, chunk_index, parent_header, content FROM chunks
             WHERE file_path = ?1 AND chunk_index BETWEEN ?2 AND ?3
             ORDER BY chunk_index"
        )?;
        let rows = stmt.query_map(params![file_path, chunk_index - w, chunk_index + w], |r| {
            Ok(Chunk {
                id: Some(r.get(0)?),
                file_path: r.get(1)?,
                chunk_index: r.get(2)?,
                parent_header: r.get(3)?,
                content: r.get(4)?,
            })
        })?;
        rows.collect::<SqliteResult<Vec<_>>>().map_err(DbError::Sqlite)
    }

    /// List all file paths in file_cache.
    pub fn list_cached_paths(&self) -> Result<Vec<String>, DbError> {
        let conn = self.write_conn.borrow();
        let mut stmt = conn.prepare("SELECT path FROM file_cache")?;
        let rows = stmt.query_map([], |r| r.get(0))?;
        rows.collect::<SqliteResult<Vec<_>>>().map_err(DbError::Sqlite)
    }

    /// Vault statistics.
    pub fn stats(&self) -> Result<VaultStats, DbError> {
        let conn = self.write_conn.borrow();
        let total_chunks: usize = conn.query_row("SELECT COUNT(*) FROM chunks", [], |r| r.get(0))?;
        let total_files: usize = conn.query_row("SELECT COUNT(*) FROM file_cache", [], |r| r.get(0))?;
        let vec_indexed: usize = conn.query_row("SELECT COUNT(*) FROM vec_chunks", [], |r| r.get(0))?;
        let total_size: usize = conn.query_row(
            "SELECT page_count * page_size FROM pragma_page_count(), pragma_page_size()",
            [], |r| r.get(0),
        ).unwrap_or(0);
        let last_indexed: Option<i64> = conn.query_row(
            "SELECT MAX(mtime) FROM file_cache", [], |r| r.get(0)
        ).ok();
        let db_path = conn.path().map(PathBuf::from).unwrap_or_default();

        Ok(VaultStats {
            total_chunks,
            total_files,
            total_size_bytes: total_size,
            last_indexed_at: last_indexed,
            db_path,
            vec_indexed_chunks: vec_indexed,
            embedder_status: String::new(), // filled by caller
        })
    }
}
```

use crate::models::Chunk;

const LEVEL2_SPLIT_THRESHOLD: usize = 1000; // chars

/// Split a Markdown document into chunks using recursive header/paragraph splitting.
/// Level 1: split on headers (#, ##, ###).
/// Level 2: if a section exceeds LEVEL2_SPLIT_THRESHOLD chars, split on blank lines.
pub fn chunk_markdown(file_path: &str, content: &str) -> Vec<Chunk> {
    if content.trim().is_empty() {
        return vec![Chunk {
            id: None,
            file_path: file_path.to_string(),
            chunk_index: 0,
            parent_header: None,
            content: String::new(),
        }];
    }

    let sections = split_by_headers(content);
    let mut chunks: Vec<Chunk> = Vec::new();

    for (header_stack, section_text) in sections {
        let parent_header = if header_stack.is_empty() {
            None
        } else {
            Some(header_stack.join(" > "))
        };

        if section_text.chars().count() > LEVEL2_SPLIT_THRESHOLD {
            // Level 2: split on blank lines (paragraph boundaries)
            let paragraphs = split_on_blank_lines(&section_text);
            for para in paragraphs {
                if !para.trim().is_empty() {
                    let idx = chunks.len() as i64;
                    chunks.push(Chunk {
                        id: None,
                        file_path: file_path.to_string(),
                        chunk_index: idx,
                        parent_header: parent_header.clone(),
                        content: para.trim().to_string(),
                    });
                }
            }
        } else if !section_text.trim().is_empty() {
            let idx = chunks.len() as i64;
            chunks.push(Chunk {
                id: None,
                file_path: file_path.to_string(),
                chunk_index: idx,
                parent_header: parent_header.clone(),
                content: section_text.trim().to_string(),
            });
        }
    }

    if chunks.is_empty() {
        chunks.push(Chunk {
            id: None,
            file_path: file_path.to_string(),
            chunk_index: 0,
            parent_header: None,
            content: content.trim().to_string(),
        });
    }

    chunks
}

/// Split content on Markdown headers (#, ##, ###).
/// Returns (header_stack, section_body) for each section.
fn split_by_headers(content: &str) -> Vec<(Vec<String>, String)> {
    let mut sections: Vec<(Vec<String>, String)> = Vec::new();
    let mut current_headers: Vec<(usize, String)> = Vec::new(); // (level, text)
    let mut current_body = String::new();
    let mut in_code_block = false;

    for line in content.lines() {
        // Track fenced code blocks to avoid splitting inside them
        if line.starts_with("```") || line.starts_with("~~~") {
            in_code_block = !in_code_block;
            current_body.push_str(line);
            current_body.push('\n');
            continue;
        }
        if in_code_block {
            current_body.push_str(line);
            current_body.push('\n');
            continue;
        }

        if let Some(level) = header_level(line) {
            if !current_body.trim().is_empty() || !current_headers.is_empty() {
                let stack: Vec<String> = current_headers.iter().map(|(_, t)| t.clone()).collect();
                sections.push((stack, current_body.clone()));
                current_body.clear();
            }
            // Pop headers at same or deeper level
            current_headers.retain(|(l, _)| *l < level);
            let title = line.trim_start_matches('#').trim().to_string();
            current_headers.push((level, title));
        } else {
            current_body.push_str(line);
            current_body.push('\n');
        }
    }

    // Push final section
    let stack: Vec<String> = current_headers.iter().map(|(_, t)| t.clone()).collect();
    sections.push((stack, current_body));

    sections
}

fn header_level(line: &str) -> Option<usize> {
    if !line.starts_with('#') { return None; }
    let count = line.chars().take_while(|c| *c == '#').count();
    if count <= 3 && line.chars().nth(count) == Some(' ') {
        Some(count)
    } else {
        None
    }
}

fn split_on_blank_lines(text: &str) -> Vec<String> {
    let mut result = Vec::new();
    let mut current = String::new();
    for line in text.lines() {
        if line.trim().is_empty() && !current.trim().is_empty() {
            result.push(current.clone());
            current.clear();
        } else {
            current.push_str(line);
            current.push('\n');
        }
    }
    if !current.trim().is_empty() {
        result.push(current);
    }
    result
}

```rust
#[derive(Error, Debug)]
pub enum EmbedderError {
    #[error("Model file not found: {0}")]
    ModelNotFound(String),
    #[error("Tokenizer file not found: {0}")]
    TokenizerNotFound(String),
    #[error("ORT error: {0}")]
    Ort(String),
    #[error("Tokenizer error: {0}")]
    Tokenizer(String),
}

pub const EMBEDDING_DIM: usize = 1024;

/// Wraps an ONNX session + HuggingFace tokenizer for Qwen3-Embedding-0.6B.
pub struct Embedder {
    session: Session,
    tokenizer: Tokenizer,
}

impl Embedder {
    /// Load model and tokenizer from the same directory.
    /// `model_path` points to `model.onnx`; `tokenizer.json` must be in the same dir.
    pub fn load<P: AsRef<Path>>(model_path: P) -> Result<Self, EmbedderError> {
        let model_path = model_path.as_ref();
        if !model_path.exists() {
            return Err(EmbedderError::ModelNotFound(
                model_path.to_string_lossy().to_string(),
            ));
        }

        let tokenizer_path = model_path
            .parent()
            .unwrap_or(Path::new("."))
            .join("tokenizer.json");
        if !tokenizer_path.exists() {
            return Err(EmbedderError::TokenizerNotFound(
                tokenizer_path.to_string_lossy().to_string(),
            ));
        }

        let session = Session::builder()
            .map_err(|e| EmbedderError::Ort(e.to_string()))?
            .commit_from_file(model_path)
            .map_err(|e| EmbedderError::Ort(e.to_string()))?;

        let tokenizer = Tokenizer::from_file(&tokenizer_path)
            .map_err(|e| EmbedderError::Tokenizer(e.to_string()))?;

        Ok(Self { session, tokenizer })
    }

    /// Embed a single text. Returns an L2-normalized vector of length EMBEDDING_DIM.
    pub fn embed(&self, text: &str) -> Result<Vec<f32>, EmbedderError> {
        let encoding = self.tokenizer
            .encode(text, true)
            .map_err(|e| EmbedderError::Tokenizer(e.to_string()))?;

        let input_ids: Vec<i64> = encoding.get_ids().iter().map(|&x| x as i64).collect();
        let attention_mask: Vec<i64> = encoding.get_attention_mask().iter().map(|&x| x as i64).collect();
        let token_type_ids: Vec<i64> = vec![0i64; input_ids.len()];
        let seq_len = input_ids.len();

        let input_ids_arr = Array2::from_shape_vec((1, seq_len), input_ids)
            .map_err(|e| EmbedderError::Ort(e.to_string()))?;
        let attention_mask_arr = Array2::from_shape_vec((1, seq_len), attention_mask.clone())
            .map_err(|e| EmbedderError::Ort(e.to_string()))?;
        let token_type_ids_arr = Array2::from_shape_vec((1, seq_len), token_type_ids)
            .map_err(|e| EmbedderError::Ort(e.to_string()))?;

        let outputs = self.session
            .run(inputs![
                "input_ids" => input_ids_arr,
                "attention_mask" => attention_mask_arr.clone(),
                "token_type_ids" => token_type_ids_arr,
            ])
            .map_err(|e| EmbedderError::Ort(e.to_string()))?;

        // Output: last_hidden_state [1, seq_len, 1024]
        let hidden: Array3<f32> = outputs[0]
            .try_extract_tensor::<f32>()
            .map_err(|e| EmbedderError::Ort(e.to_string()))?
            .to_owned()
            .into_dimensionality::<ndarray::Ix3>()
            .map_err(|e| EmbedderError::Ort(e.to_string()))?;

        // Mean pooling weighted by attention_mask
        let mask_f: Array2<f32> = attention_mask_arr.mapv(|x| x as f32);
        let mask_sum: f32 = mask_f.sum();
        let pooled: Array1<f32> = (hidden.index_axis(Axis(0), 0).t().dot(&mask_f.index_axis(Axis(0), 0))) / mask_sum;

        // L2 normalization
        let norm: f32 = pooled.iter().map(|x| x * x).sum::<f32>().sqrt();
        let normalized: Vec<f32> = if norm > 1e-8 {
            pooled.iter().map(|x| x / norm).collect()
        } else {
            pooled.to_vec()
        };

        Ok(normalized)
    }

    /// Embed a batch of texts using rayon for parallelism.
    pub fn embed_batch(&self, texts: &[&str]) -> Vec<Result<Vec<f32>, EmbedderError>> {
        // Note: ort Session is not Send, so batch runs sequentially here.
        // Callers using tokio should wrap in spawn_blocking.
        texts.iter().map(|t| self.embed(t)).collect()
    }
}

/// A stable identifier for the model, stored in file_cache.model_id.
/// Currently returns the filename stem.
pub fn model_id_for(model_path: &str) -> String {
    std::path::Path::new(model_path)
        .file_stem()

        .unwrap_or("unknown")

}

/// Resolve model path using the priority chain:
/// 1. CLI flag / explicit path argument
/// 2. SHIOTSUCHI_MODEL_PATH env var
/// 3. XDG: ~/.local/share/shiotsuchi/models/model.onnx
pub fn resolve_model_path(explicit: Option<&Path>) -> Option<std::path::PathBuf> {
    if let Some(p) = explicit {
        if p.exists() { return Some(p.to_path_buf()); }
    }
    if let Ok(env_path) = std::env::var("SHIOTSUCHI_MODEL_PATH") {
        let p = std::path::PathBuf::from(env_path);
        if p.exists() { return Some(p); }
    }
    if let Some(data_dir) = dirs::data_dir() {
        let xdg = data_dir.join("shiotsuchi").join("models").join("model.onnx");
        if xdg.exists() { return Some(xdg); }
    }
    None
}
```

```rust
    db: &NoteDatabase,
    tokenizer: &JapaneseTokenizer,
    query: &str,
    limit: usize,
    mode: SearchMode,
    embedder: Option<&Embedder>,
) -> Result<Vec<ChunkSearchResult>, SearchError> {
```

```rust
        return Ok(vec![]);
    }

    match mode {
        SearchMode::Fts => fts_search(db, tokenizer, query, limit),
        SearchMode::Vec => {

        }
    }

        let q = tokenizer.and_query(query);
        if q.is_empty() { simple_and_query(query) } else { q }
    };
    if fts5_query.is_empty() {
        return Ok(vec![]);
    }

    let pairs = db.fts_search(&fts5_query, limit)?;
    let ids: Vec<i64> = pairs.iter().map(|(id, _)| *id).collect();
    let chunks = db.get_chunks_by_ids(&ids)?;

    let results = pairs.iter().filter_map(|(id, score)| {
        chunks.iter().find(|c| c.id == Some(*id)).map(|c| ChunkSearchResult {
            chunk_id: *id,
            file_path: c.file_path.clone(),
            parent_header: c.parent_header.clone(),
            content: c.content.clone(),
            score: *score,
            search_mode: SearchMode::Fts,
        })
    }).collect();

    Ok(results)
}

fn vec_search(
    db: &NoteDatabase,
    embedder: &Embedder,
    query: &str,
    limit: usize,
) -> Result<Vec<ChunkSearchResult>, SearchError> {
    let embedding = embedder.embed(query)?;
    let pairs = db.vec_search(&embedding, limit)?;
    let ids: Vec<i64> = pairs.iter().map(|(id, _)| *id).collect();
    let chunks = db.get_chunks_by_ids(&ids)?;

    let results = pairs.iter().filter_map(|(id, dist)| {
        chunks.iter().find(|c| c.id == Some(*id)).map(|c| ChunkSearchResult {
            chunk_id: *id,
            file_path: c.file_path.clone(),
            parent_header: c.parent_header.clone(),
            content: c.content.clone(),
            score: 1.0 - dist, // cosine: distance → similarity
            search_mode: SearchMode::Vec,
        })
    }).collect();

    Ok(results)
}

fn hybrid_search(
    db: &NoteDatabase,
    tokenizer: &JapaneseTokenizer,
    embedder: &Embedder,
    query: &str,
    limit: usize,
) -> Result<Vec<ChunkSearchResult>, SearchError> {
    let fts_results = fts_search(db, tokenizer, query, limit * 2)?;
    let vec_results = vec_search(db, embedder, query, limit * 2)?;

    // RRF fusion
    let mut scores: std::collections::HashMap<i64, f64> = std::collections::HashMap::new();
    for (rank, r) in fts_results.iter().enumerate() {
        *scores.entry(r.chunk_id).or_default() +=
            DEFAULT_RRF_ALPHA / (DEFAULT_RRF_K + rank as f64 + 1.0);
    }
    for (rank, r) in vec_results.iter().enumerate() {
        *scores.entry(r.chunk_id).or_default() +=
            (1.0 - DEFAULT_RRF_ALPHA) / (DEFAULT_RRF_K + rank as f64 + 1.0);
    }

    let mut ranked: Vec<(i64, f64)> = scores.into_iter().collect();
    ranked.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());
    ranked.truncate(limit);

    let ids: Vec<i64> = ranked.iter().map(|(id, _)| *id).collect();
    let chunks = db.get_chunks_by_ids(&ids)?;

    let results = ranked.iter().filter_map(|(id, score)| {
        chunks.iter().find(|c| c.id == Some(*id)).map(|c| ChunkSearchResult {
            chunk_id: *id,
            file_path: c.file_path.clone(),
            parent_header: c.parent_header.clone(),
            content: c.content.clone(),
            score: *score,
            search_mode: SearchMode::Hybrid,
        })
    }).collect();

    Ok(results)
}

pub fn extract_snippet(text: &str, query: &str, max_chars: usize) -> String {
    let tokens: Vec<&str> = query.split_whitespace().collect();
    let lower = text.to_lowercase();

    let best_pos = tokens.iter()
        .filter_map(|t| lower.find(&t.to_lowercase()))
        .min();

    let pos = match best_pos {
        Some(p) => p,
        None => 0,
    };

    // Find start of the line containing pos
    let start = text[..pos].rfind('\n').map(|i| i + 1).unwrap_or(0);
    let snippet = &text[start..];

    if snippet.chars().count() <= max_chars {
        snippet.to_string()
    } else {
        snippet.chars().take(max_chars).collect::<String>() + "…"
    }
}
```

```rust
#[derive(Debug, Default)]
pub struct IndexReport {

    pub updated: usize,
    pub skipped: usize,
    pub errors: usize,
    pub deleted: usize,

/// Walk `vault_dir`, chunk and index all Markdown files.
/// If `embedder` is None, only FTS is indexed (vec skipped).
pub fn index_vault(
    db: &NoteDatabase,
    tokenizer: &JapaneseTokenizer,
    embedder: Option<&Embedder>,
    vault_dir: &Path,
    cfg: &IndexerConfig,
) -> Result<IndexReport, DbError> {
    let model_id = embedder
        .map(|_| "qwen3-embedding-0.6b")
        .unwrap_or("none")
        .to_string();

    let mut report = IndexReport::default();
    let mut seen_paths: std::collections::HashSet<String> = std::collections::HashSet::new();

    let mut batch: Vec<(String, Vec<crate::models::Chunk>, String, i64)> = Vec::new();

    for entry in WalkDir::new(vault_dir)
        .follow_links(false)
        .into_iter()
        .filter_entry(|e| should_visit(e, cfg))
        .filter_map(|e| e.ok())
        .filter(|e| e.file_type().is_file())
        .filter(|e| has_target_extension(e.path(), &cfg.extensions))
    {
        let abs_path = entry.path();
        let rel_path = abs_path
            .strip_prefix(vault_dir)
            .unwrap_or(abs_path)
            .to_string_lossy()
            .replace('\\', "/");

        seen_paths.insert(rel_path.clone());

        let content = match fs::read_to_string(abs_path) {
            Ok(c) => c,
            Err(e) => {
                log::warn!("Failed to read {}: {}", rel_path, e);
                report.errors += 1;
                continue;
            }
        };

        let hash = sha256_hex(&content);
        let mtime = file_mtime(abs_path);

        // Hash-based skip
        match db.cached_hash(&rel_path)? {
            Some(cached) if cached == hash => {
                report.skipped += 1;
                continue;
            }
            _ => {}
        }

        let chunks = chunk_markdown(&rel_path, &content);
        batch.push((rel_path, chunks, hash, mtime));

        if batch.len() >= cfg.batch_size {
            flush_batch(db, embedder, &model_id, &mut batch, &mut report)?;
        }
    }

    if !batch.is_empty() {
        flush_batch(db, embedder, &model_id, &mut batch, &mut report)?;
    }

    // Remove stale files (deleted from vault)
    let cached = db.list_cached_paths()?;
    for path in cached {
        if !seen_paths.contains(&path) {
            if let Err(e) = db.delete_chunks_for_file(&path) {
                log::warn!("Failed to delete stale {}: {}", path, e);
            } else {
                db.delete_file_cache(&path)?;
                report.deleted += 1;
            }
        }
    }

    Ok(report)
}

fn flush_batch(
    db: &NoteDatabase,
    embedder: Option<&Embedder>,
    model_id: &str,
    batch: &mut Vec<(String, Vec<crate::models::Chunk>, String, i64)>,
    report: &mut IndexReport,
) -> Result<(), DbError> {
    for (rel_path, chunks, hash, mtime) in batch.drain(..) {
        let is_update = db.cached_hash(&rel_path)?.is_some();

        // Delete old chunks first
        db.delete_chunks_for_file(&rel_path)?;

        // Insert new chunks
        let ids = db.insert_chunks(&chunks)?;

        // Embed and insert vectors if embedder available
        if let Some(emb) = embedder {
            let contents: Vec<&str> = chunks.iter().map(|c| c.content.as_str()).collect();
            let embeddings = emb.embed_batch(&contents);
            let pairs: Vec<(i64, Vec<f32>)> = ids.iter().zip(embeddings.iter())
                .filter_map(|(id, res)| res.as_ref().ok().map(|e| (*id, e.clone())))
                .collect();
            if let Err(e) = db.insert_embeddings(&pairs) {
                log::warn!("Failed to insert embeddings: {}", e);
            }
        }

        db.upsert_file_cache(&rel_path, &hash, mtime, model_id)?;

        if is_update {
            report.updated += 1;
        } else {
            report.inserted += 1;
        }
    }
    Ok(())
}

fn should_visit(entry: &walkdir::DirEntry, cfg: &IndexerConfig) -> bool {
    if entry.file_type().is_dir() {
        let name = entry.file_name().to_string_lossy();
        if cfg.auto_exclude_hidden && name.starts_with('.') {
            return false;
        }
        if cfg.exclude_dirs.iter().any(|d| d == name.as_ref()) {
            return false;
        }
    }
    true
}

fn has_target_extension(path: &Path, extensions: &[String]) -> bool {
    path.extension()
        .and_then(|e| e.to_str())
        .map(|e| extensions.iter().any(|ext| ext.eq_ignore_ascii_case(e)))
        .unwrap_or(false)
}

fn sha256_hex(content: &str) -> String {
    let mut hasher = Sha256::new();
    hasher.update(content.as_bytes());
    hex::encode(hasher.finalize())
}

fn file_mtime(path: &Path) -> i64 {
    fs::metadata(path)
        .and_then(|m| m.modified())
        .map(|t| t.duration_since(SystemTime::UNIX_EPOCH).unwrap_or_default().as_secs() as i64)
        .unwrap_or(0)
}
```

```rust
    let page_size: i64 = conn.query_row("PRAGMA page_size", [], |r| r.get(0))?;
    let reclaimable = freelist * page_size;

    println!("DB size       : {:.1} MB", size_before as f64 / 1_048_576.0);
    println!(
        "Reclaimable   : {:.1} MB ({} free pages)",
        reclaimable as f64 / 1_048_576.0,
        freelist
    );

    if args.analyze {
        println!("(--analyze: no VACUUM performed)");
        return Ok(());
    }

    if args.incremental {
        println!("Running incremental VACUUM...");
        conn.execute_batch("PRAGMA incremental_vacuum")?;
    } else {
        println!("Vacuuming...");
        conn.execute_batch("VACUUM")?;
```

```rust
use shiotsuchi_core::{
    db::NoteDatabase,
    embedder::{resolve_model_path, Embedder},
    indexer::{index_vault, IndexerConfig},
    tokenizer::get_tokenizer,
};
use clap::Args;
use std::path::Path;

#[derive(Args, Debug)]
pub struct ChartArgs {
    /// Force full reindex (ignore hash cache).
    #[arg(long)]
    pub force: bool,

    /// Suppress progress output.
    #[arg(long)]
    pub quiet: bool,

    /// Path to ONNX model file.
    #[arg(long)]
    pub model_path: Option<std::path::PathBuf>,
}

pub fn run_chart(
    args: &ChartArgs,
    notes_dir: &Path,
    db_path: &Path,
    _cfg: &crate::config::IndexingConfig,
) -> Result<shiotsuchi_core::indexer::IndexReport, Box<dyn std::error::Error>> {
    let db = NoteDatabase::open(db_path)?;
    let tokenizer = get_tokenizer()?;

    let embedder = match resolve_model_path(args.model_path.as_deref()) {
        Some(p) => match Embedder::load(&p) {
            Ok(e) => {
                if !args.quiet { eprintln!("[info] Embedder loaded from {}", p.display()); }
                Some(e)
            }
            Err(e) => {
                eprintln!("[warn] Could not load embedder: {}. FTS only.", e);
                None
            }
        },
        None => {
            if !args.quiet {
                eprintln!(
                    "[warn] モデルファイルが見つかりません。ベクトル検索は無効です。\n\
                     'shiotsuchi setup' を実行してモデルを配置してください。\n\
                     キーワード検索（FTS5）のみでインデックスします。"
                );
            }
            None
        }
    };

    let cfg = IndexerConfig::default();
    let report = index_vault(&db, &tokenizer, embedder.as_ref(), notes_dir, &cfg)?;

    if !args.quiet {
        eprintln!(
            "Indexed: {} inserted, {} updated, {} skipped, {} deleted, {} errors",
            report.inserted, report.updated, report.skipped, report.deleted, report.errors
        );
    }

    Ok(report)
}
```
